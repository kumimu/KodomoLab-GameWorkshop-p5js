### オプション A：敵の多様化

**概要：** 敵の種類（type）を追加し、種類ごとに動き・見た目を変える

**変更点：** 敵オブジェクト構造

```javascript
// 挿入ポイント1：グローバル変数に追加
// （CUSTOM AREA の下）

// ★ OPTION A：敵の多様化の設定
const OPTION_A_ENABLED = true;  // true にすると敵の多様化が有効

// ---

// 挿入ポイント2：敵の初期化（startGame()内）

// 敵の生成時に type を付与
for (let i = 0; i < enemyCount; i++) {
  enemies.push({
    x: random(enemySize, width - enemySize),
    y: random(-300, -50),
    speed: enemySpeed + random(0, 2),
    // ★ OPTION A：type を追加
    type: OPTION_A_ENABLED ? (random() < 0.3 ? 'fast' : 'normal') : 'normal'
  });
}

// ---

// 挿入ポイント3：敵の移動処理（moveEnemy()）

function moveEnemy(enemy) {
  // ★ OPTION A：敵のタイプに応じて速度を変更
  if (OPTION_A_ENABLED && enemy.type === 'fast') {
    enemy.y += enemy.speed * 1.5;  // 速い敵は1.5倍速
  } else {
    enemy.y += enemy.speed;
  }

  if (enemy.y > height + enemySize) {
    enemy.y = random(-300, -50);
    enemy.x = random(enemySize, width - enemySize);
    // ★ OPTION A：敵再生成時に type を再設定
    if (OPTION_A_ENABLED) {
      enemy.type = random() < 0.3 ? 'fast' : 'normal';
    }
    enemy.speed = enemySpeed + random(0, 2);
    score += 1;
  }
}

// ---

// 挿入ポイント4：敵の描画処理（drawEnemy()）

function drawEnemy(enemy) {
  fill(enemyColor);
  noStroke();
  // ★ OPTION A：敵のタイプに応じて見た目を変更
  if (OPTION_A_ENABLED && enemy.type === 'fast') {
    circle(enemy.x, enemy.y, enemySize * 0.8);  // 速い敵は小さく
  } else {
    circle(enemy.x, enemy.y, enemySize);
  }
}
```

### オプション B：射撃機能

**概要：** スペースキーで敵に向かって弾を発射

**変更点：** 新しい配列 `bullets` を追加、敵との当たり判定を拡張

```javascript
// 挿入ポイント1：グローバル変数に追加

// ★ OPTION B：射撃機能の設定
const OPTION_B_ENABLED = true;
let bullets = [];

// ---

// 挿入ポイント2：キー入力処理（keyPressed()）を拡張

function keyPressed() {
  if (gameState === "start") {
    if (key === "1") startGame(1);
    if (key === "2") startGame(2);
    if (key === "3") startGame(3);
  } else if (gameState === "end" && key === " ") {
    gameState = "start";
  }
  // ★ OPTION B：スペースキーで弾を発射
  else if (OPTION_B_ENABLED && gameState === "play" && key === " ") {
    bullets.push({
      x: playerX,
      y: playerY - playerSize / 2,
      vy: -8
    });
  }
}

// ---

// 挿入ポイント3：ゲーム中の処理（playGame()）に追加

function playGame() {
  const elapsed = floor((millis() - startTime) / 1000);
  const remaining = clearTime - elapsed;

  if (remaining <= 0) {
    resultText = "CLEAR!";
    gameState = "end";
    return;
  }

  drawGameInfo(remaining);
  movePlayer();
  drawPlayer();

  // ★ OPTION B：弾の更新と描画
  if (OPTION_B_ENABLED) {
    updateBullets();
    drawBullets();
  }

  for (const enemy of enemies) {
    moveEnemy(enemy);
    drawEnemy(enemy);

    if (isHit(enemy)) {
      resultText = "GAME OVER";
      gameState = "end";
      return;
    }
  }
}

// ---

// 挿入ポイント4：弾の処理関数を追加

function updateBullets() {
  // ★ OPTION B：弾の移動
  for (let i = bullets.length - 1; i >= 0; i--) {
    bullets[i].y += bullets[i].vy;

    // 弾が画面外に出たら削除
    if (bullets[i].y < 0) {
      bullets.splice(i, 1);
      continue;
    }

    // ★ OPTION B：敵との当たり判定
    for (let j = enemies.length - 1; j >= 0; j--) {
      const dist = hypot(
        bullets[i].x - enemies[j].x,
        bullets[i].y - enemies[j].y
      );
      if (dist < 10 + enemySize / 2) {
        // 敵に当たった
        score += 10;
        bullets.splice(i, 1);
        enemies.splice(j, 1);
        break;
      }
    }
  }
}

function drawBullets() {
  fill("yellow");
  noStroke();
  for (const bullet of bullets) {
    circle(bullet.x, bullet.y, 5);
  }
}
```

**実装難易度：★★★★** | **時間：45～60分**

### オプション H：ビジュアルカスタマイズ

**概要：** 敵を回転させたり、プレイヤーに効果を追加したりする

**変更点：** `drawEnemy()` と `drawPlayer()` に回転処理を追加

```javascript
// 挿入ポイント1：グローバル変数に追加

// ★ OPTION H：ビジュアルカスタマイズ
const OPTION_H_ENABLED = true;

// ---

// 挿入ポイント2：敵の描画処理（drawEnemy()）を拡張

function drawEnemy(enemy) {
  fill(enemyColor);
  noStroke();

  // ★ OPTION H：敵を回転させて描く
  if (OPTION_H_ENABLED) {
    push();
    translate(enemy.x, enemy.y);
    rotate(frameCount * 0.05);  // フレーム数に応じて回転
    circle(0, 0, enemySize);
    pop();
  } else {
    circle(enemy.x, enemy.y, enemySize);
  }
}

// ---

// 挿入ポイント3：プレイヤーの描画処理（drawPlayer()）を拡張

function drawPlayer() {
  fill(playerColor);
  noStroke();

  // ★ OPTION H：プレイヤーに光の効果を追加
  if (OPTION_H_ENABLED) {
    // 周りに半透明の円を描く
    fill(playerColor);
    circle(playerX, playerY, playerSize + 20);
  }

  triangle(
    playerX, playerY - playerSize / 2,
    playerX - playerSize / 2, playerY + playerSize / 2,
    playerX + playerSize / 2, playerY + playerSize / 2
  );
}
```

### オプション M：スコアシステムの工夫

**概要：** 敵のタイプごとに異なる得点、コンボシステム

**変更点：** 敵オブジェクトに `value` プロパティを追加、スコア計算ロジックを拡張

```javascript
// 挿入ポイント1：グローバル変数に追加

// ★ OPTION M：スコアシステムの工夫
const OPTION_M_ENABLED = true;
let combo = 0;
let comboTimer = 0;

// ---

// 挿入ポイント2：敵の初期化（startGame()内）に追加

for (let i = 0; i < enemyCount; i++) {
  const isRare = random() < 0.15;
  enemies.push({
    x: random(enemySize, width - enemySize),
    y: random(-300, -50),
    speed: enemySpeed + random(0, 2),
    // ★ OPTION M：敵の種類に応じて得点を設定
    value: OPTION_M_ENABLED ? (isRare ? 50 : 10) : 1
  });
}

// ---

// 挿入ポイント3：敵がクリアされた時のスコア計算（moveEnemy()）を変更

function moveEnemy(enemy) {
  enemy.y += enemy.speed;

  if (enemy.y > height + enemySize) {
    enemy.y = random(-300, -50);
    enemy.x = random(enemySize, width - enemySize);
    enemy.speed = enemySpeed + random(0, 2);

    // ★ OPTION M：敵の種類に応じて得点を加算
    if (OPTION_M_ENABLED) {
      const baseScore = enemy.value;
      const comboMultiplier = 1 + floor(combo / 5);
      score += baseScore * comboMultiplier;
      combo++;
      comboTimer = 180;  // 3秒間保持
    } else {
      score += 1;
    }
  }
}

// ---

// 挿入ポイント4：ゲーム中の処理（playGame()）にコンボ更新を追加

function playGame() {
  const elapsed = floor((millis() - startTime) / 1000);
  const remaining = clearTime - elapsed;

  if (remaining <= 0) {
    resultText = "CLEAR!";
    gameState = "end";
    return;
  }

  // ★ OPTION M：コンボタイマーを更新
  if (OPTION_M_ENABLED) {
    if (comboTimer > 0) {
      comboTimer--;
    } else {
      combo = 0;
    }
  }

  drawGameInfo(remaining);
  movePlayer();
  drawPlayer();

  for (const enemy of enemies) {
    moveEnemy(enemy);
    drawEnemy(enemy);

    if (isHit(enemy)) {
      resultText = "GAME OVER";
      gameState = "end";
      return;
    }
  }
}

// ---

// 挿入ポイント5：情報表示に追加（drawGameInfo()）

function drawGameInfo(remaining) {
  fill("white");
  textAlign(LEFT);
  textSize(18);
  text("難易度: " + difficultyName, 20, 28);
  text("残り時間: " + remaining, 20, 52);
  text("SCORE: " + score, 20, 76);

  // ★ OPTION M：コンボ表示
  if (OPTION_M_ENABLED && combo > 0) {
    fill("yellow");
    textSize(14);
    text("COMBO: " + combo + "x", 20, 100);
  }
}
```

**実装難易度：★★★** | **時間：30～45分**
